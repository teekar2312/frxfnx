   284→function step() {
   285→  for (const s of Object.values(SYMBOLS)) {
   286→    // volatilitas ~ 1 pip setiap tick + sedikit random spike untuk news simulation
   287→    const vol = s.pipSize * (0.5 + Math.random() * 1.5)
   288→    const drift = (Math.random() - 0.5) * vol * 4
   289→    s.bid = +(s.bid + drift).toFixed(s.digits)
   290→    const spread = s.pipSize * s.spreadPips
   291→    s.ask = +(s.bid + spread).toFixed(s.digits)
   292→    // simpan ke history
   293→    TICK_HISTORY[s.symbol].push({ time: Date.now(), bid: s.bid, ask: s.ask })
   294→    if (TICK_HISTORY[s.symbol].length > MAX_HISTORY) TICK_HISTORY[s.symbol].shift()
   295→  }
   296→
   297→  // emit tick
   298→  io.emit('price:tick', {
   299→    ts: Date.now(),
   300→    prices: Object.values(SYMBOLS).map((s) => ({ symbol: s.symbol, bid: s.bid, ask: s.ask, digits: s.digits })),
   301→  })
   302→
   303→  // update equity + positions
   304→  recomputeEquity()
   305→
   306→  // emit positions with profit + duration
   307→  const now = Date.now()
   308→  const posArr = Array.from(positions.values()).map((p) => ({
   309→    ...p,
   310→    bid: SYMBOLS[p.symbol]?.bid ?? 0,
   311→    ask: SYMBOLS[p.symbol]?.ask ?? 0,
   312→    profit: computeProfit(p),
   313→    durationMs: now - new Date(p.openTime).getTime(),
   314→  }))
   315→  io.emit('positions:update', posArr)
   316→
   317→  // emit pending orders
   318→  const pendArr = Array.from(pendingOrders.values())
   319→  io.emit('pending:orders:update', pendArr)
   320→
   321→  // Process pending orders (check trigger)
   322→  for (const po of pendingOrders.values()) {
   323→    if (po.status !== 'pending') continue
   324→    const s = SYMBOLS[po.symbol]
   325→    if (!s) continue
   326→    let triggered = false
   327→    if (po.orderType === 'BUY_STOP' && s.ask >= po.triggerPrice) triggered = true
   328→    if (po.orderType === 'SELL_STOP' && s.bid <= po.triggerPrice) triggered = true
   329→    if (po.orderType === 'BUY_LIMIT' && s.ask <= po.triggerPrice) triggered = true
   330→    if (po.orderType === 'SELL_LIMIT' && s.bid >= po.triggerPrice) triggered = true
   331→    if (triggered) {
   332→      po.status = 'filled'
   333→      const side = po.orderType.startsWith('BUY') ? 'BUY' : 'SELL'
   334→      const r = openMarket(po.symbol, side, po.lots, { sl: po.sl, tp: po.tp, isAuto: po.isAuto, strategyId: po.strategyId })
   335→      if (r.ok) {
   336→        logEvent('info', 'trading', `PENDING #${po.ticket} FILLED → position #${r.ticket} ${po.orderType} ${po.lots} ${po.symbol} @ ${po.triggerPrice}`)
   337→        io.emit('pending:filled', { pendingTicket: po.ticket, positionTicket: r.ticket })
   338→      }
   339→    }
   340→    // expiration check
   341→    if (po.expiresAt && new Date(po.expiresAt).getTime() < now) {
   342→      po.status = 'expired'
   343→      logEvent('info', 'trading', `PENDING #${po.ticket} EXPIRED`)
   344→    }
   345→  }
   346→
   347→  // Trailing stop
   348→  for (const p of positions.values()) {
   349→    if (!p.trailingStopPips || p.trailingStopPips <= 0) continue
   350→    const s = SYMBOLS[p.symbol]
   351→    if (!s) continue
   352→    const pipSize = s.pipSize
   353→    if (p.side === 'BUY') {
   354→      const newSL = +(s.bid - p.trailingStopPips * pipSize).toFixed(s.digits)
   355→      if (!p.sl || newSL > p.sl) {
   356→        p.sl = newSL
   357→        p.trailingActivated = true
   358→      }
   359→      // TP hit
   360→      if (p.tp && s.bid >= p.tp) {
   361→        closePosition(p.ticket, 'tp')
   362→      }
   363→      // SL hit
   364→      if (p.sl && s.bid <= p.sl) {
   365→        closePosition(p.ticket, p.trailingActivated ? 'trailing' : 'sl')
   366→      }
   367→    } else {
   368→      const newSL = +(s.ask + p.trailingStopPips * pipSize).toFixed(s.digits)
   369→      if (!p.sl || newSL < p.sl) {
   370→        p.sl = newSL
   371→        p.trailingActivated = true
   372→      }
   373→      if (p.tp && s.ask <= p.tp) closePosition(p.ticket, 'tp')
   374→      if (p.sl && s.ask >= p.sl) closePosition(p.ticket, p.trailingActivated ? 'trailing' : 'sl')
   375→    }
   376→  }
   377→
   378→  // Alerts evaluation
   379→  evaluateAlerts()
   380→  // Auto-disable pre-news
   381→  evaluateAutoTradeRules()
   382→}

   384→function computeProfit(p: Position): number {
   385→  const s = SYMBOLS[p.symbol]
   386→  if (!s) return 0
   387→  const close = p.side === 'BUY' ? s.bid : s.ask
   388→  const diff = p.side === 'BUY' ? close - p.openPrice : p.openPrice - close
   389→  return +(diff * p.lots * s.contractSize).toFixed(2)
   390→}
   391→
   392→function recomputeEquity() {
   393→  let floating = 0
   394→  for (const p of positions.values()) floating += computeProfit(p)
   395→  mt5.equity = +(mt5.balance + floating).toFixed(2)
   396→  mt5.freeMargin = +(mt5.equity - mt5.margin).toFixed(2)
   397→  mt5.marginLevel = mt5.margin > 0 ? +((mt5.equity / mt5.margin) * 100).toFixed(2) : 0
   398→}

   405→function isTradingAllowedNow(): boolean {
   406→  const now = new Date()
   407→  const day = now.getDay()
   408→  if (day === 0 || day === 6) return false
   409→  const hhmm = now.getHours() * 100 + now.getMinutes()
   410→  return hhmm >= 800 && hhmm <= 2330

   413→function openMarket(symbol: string, side: 'BUY' | 'SELL', lots: number, opts?: { sl?: number; tp?: number; isAuto?: boolean; strategyId?: string; trailingStopPips?: number; originalLots?: number }): { ok: boolean; ticket?: number; msg?: string } {
   414→  if (!mt5.connected) return { ok: false, msg: 'MT5 belum terhubung' }
   415→  const s = SYMBOLS[symbol]
   416→  if (!s) return { ok: false, msg: 'Symbol tidak ditemukan' }
   417→  if (!isTradingAllowedNow()) return { ok: false, msg: 'Di luar jam trading (atau akhir pekan)' }
   418→
   419→  // Spread filter
   420→  const spreadPipsNow = (s.ask - s.bid) / s.pipSize
   421→  if (spreadPipsNow > s.maxSpreadPips) {
   422→    logEvent('warn', 'trading', `Spread reject: ${symbol} spread ${spreadPipsNow.toFixed(1)} pips > max ${s.maxSpreadPips}`)
   423→    return { ok: false, msg: `Spread terlalu lebar (${spreadPipsNow.toFixed(1)} pips > ${s.maxSpreadPips}). Entry ditolak.` }
   424→  }
   425→
   426→  const price = side === 'BUY' ? s.ask : s.bid
   427→  const ticket = ++ticketSeq
   428→  positions.set(ticket, {
   429→    ticket,
   430→    symbol,
   431→    side,
   432→    lots,
   433→    originalLots: opts?.originalLots ?? lots,
   434→    openPrice: price,
   435→    sl: opts?.sl,
   436→    tp: opts?.tp,
   437→    openTime: new Date().toISOString(),
   438→    strategyId: opts?.strategyId,
   439→    isAuto: opts?.isAuto ?? false,
   440→    trailingStopPips: opts?.trailingStopPips,
   441→  })
   442→  const notional = lots * s.contractSize
   443→  const usedMargin = notional / mt5.leverage
   444→  mt5.margin = +(mt5.margin + usedMargin).toFixed(2)
   445→  logEvent('info', 'trading', `OPEN ${side} ${lots} ${symbol} @ ${price}`, { ticket, sl: opts?.sl, tp: opts?.tp, isAuto: opts?.isAuto })
   446→  return { ok: true, ticket }
   447→}

   449→function closePosition(ticket: number, reason: TradeHistory['closeReason'] = 'manual'): { ok: boolean; msg?: string; profit?: number } {
   450→  const p = positions.get(ticket)
   451→  if (!p) return { ok: false, msg: 'Posisi tidak ditemukan' }
   452→  const profit = computeProfit(p)
   453→  mt5.balance = +(mt5.balance + profit).toFixed(2)
   454→  const s = SYMBOLS[p.symbol]
   455→  const notional = p.lots * s.contractSize
   456→  const usedMargin = notional / mt5.leverage
   457→  mt5.margin = +(mt5.margin - usedMargin).toFixed(2)
   458→  const closePrice = p.side === 'BUY' ? s.bid : s.ask
   459→  const pnlPips = +((p.side === 'BUY' ? closePrice - p.openPrice : p.openPrice - closePrice) / s.pipSize).toFixed(1)
   460→  const durationMs = Date.now() - new Date(p.openTime).getTime()
   461→  tradeHistory.unshift({
   462→    ticket: p.ticket,
   463→    symbol: p.symbol,
   464→    side: p.side,
   465→    lots: p.lots,
   466→    openPrice: p.openPrice,
   467→    closePrice,
   468→    openTime: p.openTime,
   469→    closeTime: new Date().toISOString(),
   470→    profit,
   471→    pnlPips,
   472→    strategyId: p.strategyId,
   473→    isAuto: p.isAuto,
   474→    closeReason: reason,
   475→    durationMs,
   476→  })
   477→  // keep last 500 trades
   478→  if (tradeHistory.length > 500) tradeHistory.length = 500
   479→  logEvent('info', 'trading', `CLOSE #${ticket} ${p.symbol} P&L=${profit} (${reason})`, { ticket, profit, reason })
   480→  positions.delete(ticket)
   481→  return { ok: true, profit }
   482→}

   484→function partialClose(ticket: number, fraction: number): { ok: boolean; msg?: string; profit?: number; remainingLots?: number } {
   485→  const p = positions.get(ticket)
   486→  if (!p) return { ok: false, msg: 'Posisi tidak ditemukan' }
   487→  if (fraction <= 0 || fraction >= 1) return { ok: false, msg: 'Fraction harus 0 < f < 1' }
   488→  const closeLots = +(p.lots * fraction).toFixed(2)
   489→  if (closeLots < 0.01) return { ok: false, msg: 'Lot tersisa terlalu kecil' }
   490→  const s = SYMBOLS[p.symbol]
   491→  const closePrice = p.side === 'BUY' ? s.bid : s.ask
   492→  const profit = +((p.side === 'BUY' ? closePrice - p.openPrice : p.openPrice - closePrice) * closeLots * s.contractSize).toFixed(2)
   493→  mt5.balance = +(mt5.balance + profit).toFixed(2)
   494→  // kurangi margin
   495→  const oldNotional = p.lots * s.contractSize
   496→  const newLots = +(p.lots - closeLots).toFixed(2)
   497→  const newNotional = newLots * s.contractSize
   498→  const marginDiff = (oldNotional - newNotional) / mt5.leverage
   499→  mt5.margin = +(mt5.margin - marginDiff).toFixed(2)
   500→  p.lots = newLots
   501→  p.partialClosed = true
   502→  logEvent('info', 'trading', `PARTIAL CLOSE #${ticket} ${fraction * 100}% P&L=${profit}`, { ticket, profit, remainingLots: newLots })
   503→  return { ok: true, profit, remainingLots: newLots }
   504→}

   506→function modifyPosition(ticket: number, sl?: number, tp?: number, trailingStopPips?: number) {
   507→  const p = positions.get(ticket)
   508→  if (!p) return { ok: false, msg: 'Posisi tidak ditemukan' }
   509→  if (sl !== undefined) p.sl = sl
   510→  if (tp !== undefined) p.tp = tp
   511→  if (trailingStopPips !== undefined) p.trailingStopPips = trailingStopPips
   512→  logEvent('info', 'trading', `MODIFY #${ticket} SL=${sl} TP=${tp} Trail=${trailingStopPips}`)
   513→  return { ok: true }
   514→}

   516→// ============================================================
   517→// PENDING ORDERS
   518→// ============================================================
   519→function placePendingOrder(p: {
   520→  symbol: string
   521→  orderType: 'BUY_STOP' | 'SELL_STOP' | 'BUY_LIMIT' | 'SELL_LIMIT'
   522→  lots: number
   523→  triggerPrice: number
   524→  sl?: number
   525→  tp?: number
   526→  isAuto?: boolean
   527→  strategyId?: string
   528→  expiresAt?: string
   529→}): { ok: boolean; ticket?: number; msg?: string } {
   530→  if (!mt5.connected) return { ok: false, msg: 'MT5 belum terhubung' }
   531→  const s = SYMBOLS[p.symbol]
   532→  if (!s) return { ok: false, msg: 'Symbol tidak ditemukan' }
   533→  if (!isTradingAllowedNow()) return { ok: false, msg: 'Di luar jam trading' }
   534→  // Validasi: BUY_STOP harus di atas ask, SELL_STOP di bawah bid, dst
   535→  if (p.orderType === 'BUY_STOP' && p.triggerPrice <= s.ask) return { ok: false, msg: 'BUY_STOP harus di atas ask saat ini' }
   536→  if (p.orderType === 'SELL_STOP' && p.triggerPrice >= s.bid) return { ok: false, msg: 'SELL_STOP harus di bawah bid saat ini' }
   537→  if (p.orderType === 'BUY_LIMIT' && p.triggerPrice >= s.ask) return { ok: false, msg: 'BUY_LIMIT harus di bawah ask saat ini' }
   538→  if (p.orderType === 'SELL_LIMIT' && p.triggerPrice <= s.bid) return { ok: false, msg: 'SELL_LIMIT harus di atas bid saat ini' }
   539→  const ticket = ++pendingSeq
   540→  pendingOrders.set(ticket, {
   541→    ticket,
   542→    symbol: p.symbol,
   543→    orderType: p.orderType,
   544→    lots: p.lots,
   545→    triggerPrice: p.triggerPrice,
   546→    sl: p.sl,
   547→    tp: p.tp,
   548→    status: 'pending',
   549→    strategyId: p.strategyId,
   550→    isAuto: p.isAuto ?? false,
   551→    expiresAt: p.expiresAt,
   552→    createdAt: new Date().toISOString(),
   553→  })
   554→  logEvent('info', 'trading', `PENDING ${p.orderType} ${p.lots} ${p.symbol} @ ${p.triggerPrice}`, { ticket })
   555→  return { ok: true, ticket }
   556→}
   557→
   558→function cancelPendingOrder(ticket: number) {
   559→  const po = pendingOrders.get(ticket)
   560→  if (!po) return { ok: false, msg: 'Pending order tidak ditemukan' }
   561→  po.status = 'cancelled'
   562→  logEvent('info', 'trading', `PENDING #${ticket} CANCELLED`)
   563→  return { ok: true }
   564→}

   566→// ============================================================
   567→// ALERTS ENGINE
   568→// ============================================================
   569→function evaluateAlerts() {
   570→  const now = new Date()
   571→  for (const rule of alertRules) {
   572→    if (!rule.enabled) continue
   573→    let triggered = false
   574→    let symbol: string | undefined
   575→    if (rule.type === 'daily_stop') {
   576→      // today's P&L
   577→      const today = now.toISOString().slice(0, 10)
   578→      if (sessionStartBalance.date !== today) {
   579→        sessionStartBalance.date = today
   580→        sessionStartBalance.balance = mt5.balance
   581→      }
   582→      const dayPnl = mt5.balance - sessionStartBalance.balance
   583→      const dayPnlPct = (dayPnl / sessionStartBalance.balance) * 100
   584→      if (dayPnlPct <= -(rule.condition.pct ?? 3)) triggered = true
   585→    } else if (rule.type === 'margin_low') {
   586→      if (mt5.marginLevel > 0 && mt5.marginLevel < (rule.condition.pct ?? 50)) triggered = true
   587→    } else if (rule.type === 'spread_wide') {
   588→      for (const s of Object.values(SYMBOLS)) {
   589→        const spreadPips = (s.ask - s.bid) / s.pipSize
   590→        if (spreadPips > (rule.condition.pips ?? 5)) {
   591→          triggered = true
   592→          symbol = s.symbol
   593→          break
   594→        }
   595→      }
   596→    } else if (rule.type === 'price_above' || rule.type === 'price_below') {
   597→      if (rule.symbol) {
   598→        const s = SYMBOLS[rule.symbol]
   599→        if (s) {
   600→          if (rule.type === 'price_above' && s.bid >= rule.condition.price) triggered = true
   601→          if (rule.type === 'price_below' && s.bid <= rule.condition.price) triggered = true
   602→        }
   603→      }
   604→    }
   605→    if (triggered) {
   606→      // cooldown 60s
   607→      if (rule.lastTriggeredAt && Date.now() - new Date(rule.lastTriggeredAt).getTime() < 60000) continue
   608→      rule.triggerCount++
   609→      rule.lastTriggeredAt = now.toISOString()
   610→      const msg = rule.message.replace('{symbol}', symbol ?? rule.symbol ?? '')
   611→      const ev: AlertEvent = {
   612→        id: 'alert_' + Math.random().toString(36).slice(2),
   613→        ruleId: rule.id,
   614→        type: rule.type,
   615→        message: msg,
   616→        severity: rule.severity,
   617→        timestamp: now.toISOString(),
   618→        acknowledged: false,
   619→      }
   620→      alertEvents.unshift(ev)
   621→      if (alertEvents.length > 100) alertEvents.length = 100
   622→      io.emit('alert:event', ev)
   623→      logEvent(rule.severity === 'critical' ? 'error' : 'warn', 'alert', msg, { ruleId: rule.id })
   624→    }
   625→  }
   626→  io.emit('alerts:list', alertEvents.slice(0, 30))
   627→}

   629→// ============================================================
   630→// AUTO-TRADE RULES (pre-news flatten, max daily loss, etc)
   631→// ============================================================
   632→function evaluateAutoTradeRules() {
   633→  for (const rule of autoTradeRules) {
   634→    if (!rule.enabled) continue
   635→    if (rule.type === 'max_daily_loss') {
   636→      const today = new Date().toISOString().slice(0, 10)
   637→      if (sessionStartBalance.date !== today) {
   638→        sessionStartBalance.date = today
   639→        sessionStartBalance.balance = mt5.balance
   640→      }
   641→      const dayPnl = mt5.balance - sessionStartBalance.balance
   642→      const dayPnlPct = (dayPnl / sessionStartBalance.balance) * 100
   643→      if (dayPnlPct <= -3) {
   644→        // flatten all
   645→        flattenAll('max_daily_loss')
   646→        rule.enabled = false  // disable setelah trigger
   647→      }
   648→    } else if (rule.type === 'pre_news_flatten') {
   649→      // cek apakah ada berita high impact dalam X menit
   650→      const minutesBefore = rule.minutesBefore ?? 2
   651→      const nowMs = Date.now()
   652→      for (const n of NEWS) {
   653→        if (n.impact !== (rule.impactFilter ?? 'high')) continue
   654→        if (rule.countryFilter && n.country !== rule.countryFilter) continue
   655→        const timeToNews = new Date(n.eventTime).getTime() - nowMs
   656→        if (timeToNews > 0 && timeToNews < minutesBefore * 60 * 1000) {
   657→          flattenAll('pre_news_flatten')
   658→          logEvent('warn', 'strategy', `Auto-flatten pre-news: ${n.title} in ${Math.round(timeToNews / 60000)}min`)
   659→          break
   660→        }
   661→      }
   662→    }
   663→  }
   664→}
   665→
   666→function flattenAll(reason: string): number {
   667→  let closed = 0
   668→  for (const ticket of Array.from(positions.keys())) {
   669→    closePosition(ticket, reason as any)
   670→    closed++
   671→  }
   672→  if (closed > 0) logEvent('warn', 'strategy', `FLATTEN ALL (${reason}): ${closed} positions closed`)
   673→  return closed
   674→}

   679→interface NewsEvent {
   680→  id: string
   681→  title: string
   682→  country: string
   683→  currency: string
   684→  impact: 'high' | 'medium' | 'low'
   685→  eventTime: string
   686→  previous?: number
   687→  forecast?: number
   688→  actual?: number
   689→  source: string
   690→}
   691→const NEWS: NewsEvent[] = [
   692→  { id: 'n1', title: 'CPI m/m (US)', country: 'US', currency: 'USD', impact: 'high', eventTime: new Date(Date.now() + 3600 * 1000).toISOString(), previous: 2.4, forecast: 2.4, source: 'tiingo' },

   695→  { id: 'n4', title: 'GDP q/q', country: 'US', currency: 'USD', impact: 'high', eventTime: new Date(Date.now() + 21600 * 1000).toISOString(), previous: 3.0, forecast: 2.8, source: 'tiingo' },

   698→function analyzeNews(ev: NewsEvent) {
   699→  if (ev.actual === undefined || ev.forecast === undefined) return { ready: false, deviation: null, directionUsd: 'neutral' }
   700→  const deviation = ev.forecast !== 0 ? +(((ev.actual - ev.forecast) / Math.abs(ev.forecast)) * 100).toFixed(2) : 0
   701→  const actualGreater = ev.actual > ev.forecast
   702→  const title = ev.title.toLowerCase()
   703→  let directionUsd: 'up' | 'down' | 'neutral' = 'neutral'
   704→  if (title.includes('unemployment')) directionUsd = actualGreater ? 'down' : 'up'
   705→  else directionUsd = actualGreater ? 'up' : 'down'
   706→  const invert = (d: string) => (d === 'up' ? 'down' : d === 'down' ? 'up' : 'neutral')
   707→  return {
   708→    ready: true,
   709→    deviation,
   710→    directionUsd,
   711→    directionXau: invert(directionUsd),
   712→    directionEur: invert(directionUsd),
   713→    directionGbp: invert(directionUsd),
   714→    directionJpy: directionUsd,
   715→  }
   716→}

   884→    const symbol = cfg.symbol
   885→    const history = TICK_HISTORY[symbol] ?? []
   886→    if (history.length < 50) {
   887→      return socket.emit('backtest:result', { ok: false, msg: 'History tidak cukup (min 50 ticks)' })
   888→    }
   889→    // Simple strategy: BUY when price > SMA(20), SELL when < SMA(20)
   890→    let balance = cfg.initialBalance ?? 10000
   891→    let position: any = null
   892→    const trades: any[] = []
   893→    const smaPeriod = cfg.smaPeriod ?? 20
   894→    for (let i = smaPeriod; i < history.length; i++) {
   895→      const slice = history.slice(i - smaPeriod, i)
   896→      const sma = slice.reduce((a, t) => a + t.bid, 0) / slice.length
   897→      const cur = history[i]
   898→      if (!position && cur.bid > sma) {
   899→        position = { side: 'BUY', entry: cur.bid, time: cur.time, lots: cfg.lots ?? 0.1 }
   900→      } else if (position && position.side === 'BUY' && cur.bid < sma) {
   901→        const profit = (cur.bid - position.entry) * position.lots * (SYMBOLS[symbol]?.contractSize ?? 100000)
   902→        balance += profit
   903→        trades.push({ ...position, exit: cur.bid, exitTime: cur.time, profit, pnlPips: (cur.bid - position.entry) / (SYMBOLS[symbol]?.pipSize ?? 0.0001) })
   904→        position = null
   905→      } else if (!position && cur.bid < sma) {
   906→        position = { side: 'SELL', entry: cur.bid, time: cur.time, lots: cfg.lots ?? 0.1 }
   907→      } else if (position && position.side === 'SELL' && cur.bid > sma) {
   908→        const profit = (position.entry - cur.bid) * position.lots * (SYMBOLS[symbol]?.contractSize ?? 100000)
   909→        balance += profit
   910→        trades.push({ ...position, exit: cur.bid, exitTime: cur.time, profit, pnlPips: (position.entry - cur.bid) / (SYMBOLS[symbol]?.pipSize ?? 0.0001) })
   911→        position = null
   912→      }
   913→    }
   914→    const wins = trades.filter((t) => t.profit > 0).length
   915→    const losses = trades.filter((t) => t.profit <= 0).length
   916→    const grossProfit = trades.filter((t) => t.profit > 0).reduce((a, t) => a + t.profit, 0)
   917→    const grossLoss = Math.abs(trades.filter((t) => t.profit <= 0).reduce((a, t) => a + t.profit, 0))
   918→    const winRate = trades.length > 0 ? (wins / trades.length) * 100 : 0
   919→    const profitFactor = grossLoss > 0 ? grossProfit / grossLoss : 0
   920→    // Max drawdown
   921→    let peak = balance, maxDD = 0
   922→    let runningBal = cfg.initialBalance ?? 10000
   923→    for (const t of trades) {
   924→      runningBal += t.profit
   925→      peak = Math.max(peak, runningBal)
   926→      maxDD = Math.max(maxDD, ((peak - runningBal) / peak) * 100)
   927→    }
   928→    socket.emit('backtest:result', {
   929→      ok: true,
   930→      totalTrades: trades.length,
   931→      wins,
   932→      losses,
   933→      winRate: +winRate.toFixed(2),
   934→      profitFactor: +profitFactor.toFixed(2),
   935→      maxDrawdown: +maxDD.toFixed(2),
   936→      finalBalance: +balance.toFixed(2),
   937→      trades: trades.slice(-100),
   938→    })
   939→    logEvent('info', 'backtest', `Backtest ${symbol}: ${trades.length} trades, win rate ${winRate.toFixed(1)}%`, { balance })
   940→  })
